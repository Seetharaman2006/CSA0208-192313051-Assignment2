#include <stdio.h>
#include <stdint.h>

#define MAX_READING 65535U
#define FIXED_CHARGE_PAISE 5000LL
#define RATE2_PAISE 225LL
#define RATE3_PAISE 450LL

#ifndef DEBUG
#define DEBUG 0
#endif

#if DEBUG
#define DEBUG_PRINT(...) printf("[DEBUG] "); printf(__VA_ARGS__)
#else
#define DEBUG_PRINT(...) ((void)0)
#endif

/* Function prototypes */
uint32_t calculate_units(uint32_t previous, uint32_t current);
int64_t calculate_bill(uint32_t units);
void print_money(int64_t paise);

/* Calculate units, including meter rollover */
uint32_t calculate_units(uint32_t previous, uint32_t current)
{
    uint32_t units;

    if (current >= previous) {
        units = current - previous;
    } else {
        units = (MAX_READING - previous) + 1U + current;
    }

    DEBUG_PRINT("Previous=%u, Current=%u, Units=%u\n",
                previous, current, units);

    return units;
}

/* Calculate electricity bill in paise */
int64_t calculate_bill(uint32_t units)
{
    int64_t energy_charge = 0;

    if (units > 100U) {
        uint32_t second_slab = units - 100U;

        if (second_slab > 100U) {
            second_slab = 100U;
        }

        energy_charge += (int64_t)second_slab * RATE2_PAISE;
    }

    if (units > 200U) {
        energy_charge +=
            (int64_t)(units - 200U) * RATE3_PAISE;
    }

    return FIXED_CHARGE_PAISE + energy_charge;
}

/* Display rupees and paise */
void print_money(int64_t paise)
{
    printf("Rs %lld.%02lld",
           (long long)(paise / 100),
           (long long)(paise % 100));
}

int main(void)
{
    uint32_t units;
    int64_t bill;

    printf("TASK 1: HEADER FILES AND CONDITIONAL COMPILATION\n\n");

    units = calculate_units(1200U, 1530U);
    bill = calculate_bill(units);

    printf("Normal meter reading:\n");
    printf("Units consumed: %u\n", units);
    printf("Total bill: ");
    print_money(bill);
    printf("\n\n");

    units = calculate_units(65400U, 120U);
    bill = calculate_bill(units);

    printf("Meter rollover:\n");
    printf("Units consumed: %u\n", units);
    printf("Total bill: ");
    print_money(bill);
    printf("\n");

    return 0;
}
